function crearError(codigo, mensaje) {
  const error = new Error(mensaje);
  error.code = codigo;
  return error;
}

function obtenerUsuario() {
  const usuario = auth.currentUser;

  if (!usuario || usuario.isAnonymous) {
    throw crearError(
      "asistencia/no-autenticado",
      "Debes iniciar sesión con tu cuenta de empleado."
    );
  }

  return usuario;
}

function obtenerFechaActual() {
  return new Date();
}

function formatearFecha(fecha, opciones = {}) {
  const valor = convertirFecha(fecha);

  if (!valor) {
    return "—";
  }

  return new Intl.DateTimeFormat("es-MX", {
    timeZone: ZONA_HORARIA,
    day: "2-digit",
    month: "long",
    year: "numeric",
    ...opciones
  }).format(valor);
}

function formatearHora(fecha, segundos = false) {
  const valor = convertirFecha(fecha);

  if (!valor) {
    return "—";
  }

  return new Intl.DateTimeFormat("es-MX", {
    timeZone: ZONA_HORARIA,
    hour: "2-digit",
    minute: "2-digit",
    ...(segundos ? { second: "2-digit" } : {}),
    hour12: false
  }).format(valor);
}

function obtenerClaveFecha(fecha = new Date()) {
  const valor = convertirFecha(fecha);

  if (!valor) {
    return "";
  }

  const partes = new Intl.DateTimeFormat("en-CA", {
    timeZone: ZONA_HORARIA,
    year: "numeric",
    month: "2-digit",
    day: "2-digit"
  }).formatToParts(valor);

  const datos = {};

  partes.forEach(parte => {
    if (parte.type !== "literal") {
      datos[parte.type] = parte.value;
    }
  });

  return `${datos.year}-${datos.month}-${datos.day}`;
}

function obtenerClaveMes(fecha = new Date()) {
  return obtenerClaveFecha(fecha).slice(0, 7);
}

function convertirFecha(valor) {
  if (!valor) {
    return null;
  }

  if (valor instanceof Date) {
    return Number.isNaN(valor.getTime()) ? null : valor;
  }

  if (valor instanceof Timestamp) {
    return valor.toDate();
  }

  if (typeof valor?.toDate === "function") {
    return valor.toDate();
  }

  if (typeof valor === "number") {
    const fecha = new Date(valor);
    return Number.isNaN(fecha.getTime()) ? null : fecha;
  }

  if (typeof valor === "string") {
    const fecha = new Date(valor);
    return Number.isNaN(fecha.getTime()) ? null : fecha;
  }

  if (typeof valor?.seconds === "number") {
    return new Date(
      valor.seconds * 1000 +
      Math.floor((valor.nanoseconds || 0) / 1000000)
    );
  }

  return null;
}

function validarHora(hora) {
  if (typeof hora !== "string") {
    return false;
  }

  if (!/^([01]\d|2[0-3]):[0-5]\d$/.test(hora)) {
    return false;
  }

  return true;
}

function horaAMinutos(hora) {
  if (!validarHora(hora)) {
    throw crearError(
      "horario/invalido",
      "El horario debe tener el formato HH:MM."
    );
  }

  const [horas, minutos] = hora.split(":").map(Number);

  return horas * 60 + minutos;
}

function calcularDuracionTurno(entrada, salida) {
  const inicio = horaAMinutos(entrada);
  let fin = horaAMinutos(salida);

  if (fin <= inicio) {
    fin += 1440;
  }

  return fin - inicio;
}

function normalizarConfiguracion(datos = {}) {
  const configuracion = {
    ...CONFIGURACION_PREDETERMINADA,
    ...datos
  };

  const enteros = [
    "toleranciaMinutos",
    "limiteEntradaMinutos",
    "radioMetros",
    "precisionMaximaMetros"
  ];

  enteros.forEach(campo => {
    configuracion[campo] = Number(configuracion[campo]);
  });

  return configuracion;
}

function validarConfiguracion(configuracion) {
  const datos = normalizarConfiguracion(configuracion);

  if (
    !Number.isInteger(datos.toleranciaMinutos) ||
    datos.toleranciaMinutos < 0 ||
    datos.toleranciaMinutos > 1440
  ) {
    throw crearError(
      "configuracion/tolerancia-invalida",
      "La tolerancia debe estar entre 0 y 1440 minutos."
    );
  }

  if (
    !Number.isInteger(datos.limiteEntradaMinutos) ||
    datos.limiteEntradaMinutos < datos.toleranciaMinutos ||
    datos.limiteEntradaMinutos > 1440
  ) {
    throw crearError(
      "configuracion/limite-invalido",
      "El límite de entrada debe ser igual o mayor que la tolerancia."
    );
  }

  if (
    !Number.isFinite(datos.radioMetros) ||
    datos.radioMetros < 10 ||
    datos.radioMetros > 10000
  ) {
    throw crearError(
      "configuracion/radio-invalido",
      "El radio permitido debe estar entre 10 y 10000 metros."
    );
  }

  if (
    !Number.isFinite(datos.precisionMaximaMetros) ||
    datos.precisionMaximaMetros < 5 ||
    datos.precisionMaximaMetros > 1000
  ) {
    throw crearError(
      "configuracion/precision-invalida",
      "La precisión máxima debe estar entre 5 y 1000 metros."
    );
  }

  return datos;
}

async function obtenerConfiguracionGeneral() {
  const referencia = doc(db, "configuracion", "asistencias");
  const documento = await getDoc(referencia);

  if (!documento.exists()) {
    throw crearError(
      "configuracion/no-configurada",
      "El administrador todavía no ha configurado el control de asistencias."
    );
  }

  return validarConfiguracion(documento.data());
}

async function obtenerSucursal(sucursalId) {
  if (!sucursalId) {
    throw crearError(
      "sucursal/no-asignada",
      "Tu perfil no tiene una sucursal asignada."
    );
  }

  const referencia = doc(db, "sucursales", sucursalId);
  const documento = await getDoc(referencia);

  if (!documento.exists()) {
    throw crearError(
      "sucursal/no-encontrada",
      "No encontramos la sucursal asignada."
    );
  }

  const datos = documento.data();

  if (datos.activo !== true) {
    throw crearError(
      "sucursal/inactiva",
      "La sucursal asignada no está activa."
    );
  }

  return {
    id: documento.id,
    ...datos
  };
}

function gradosARadianes(grados) {
  return grados * Math.PI / 180;
}

function calcularDistanciaMetros(lat1, lon1, lat2, lon2) {
  const radioTierra = 6371000;

  const diferenciaLatitud = gradosARadianes(lat2 - lat1);
  const diferenciaLongitud = gradosARadianes(lon2 - lon1);

  const a =
    Math.sin(diferenciaLatitud / 2) ** 2 +
    Math.cos(gradosARadianes(lat1)) *
    Math.cos(gradosARadianes(lat2)) *
    Math.sin(diferenciaLongitud / 2) ** 2;

  const valor = Math.min(1, Math.max(0, a));

  return 2 * radioTierra * Math.atan2(
    Math.sqrt(valor),
    Math.sqrt(1 - valor)
  );
}

function validarCoordenadas(latitud, longitud) {
  return (
    Number.isFinite(latitud) &&
    Number.isFinite(longitud) &&
    latitud >= -90 &&
    latitud <= 90 &&
    longitud >= -180 &&
    longitud <= 180
  );
}

function verificarGeocerca(ubicacion, sucursal, configuracion) {
  const latitud = Number(ubicacion?.latitud);
  const longitud = Number(ubicacion?.longitud);
  const precision = Number(ubicacion?.precision);

  const latitudSucursal = Number(sucursal?.latitud);
  const longitudSucursal = Number(sucursal?.longitud);

  if (
    !validarCoordenadas(latitud, longitud) ||
    !validarCoordenadas(latitudSucursal, longitudSucursal)
  ) {
    throw crearError(
      "ubicacion/coordenadas-invalidas",
      "No se pudieron verificar las coordenadas."
    );
  }

  if (!Number.isFinite(precision) || precision < 0) {
    throw crearError(
      "ubicacion/precision-invalida",
      "El dispositivo no proporcionó una precisión válida."
    );
  }

  const radioMetros = Number(
    sucursal.radioMetros ?? configuracion.radioMetros
  );

  const distanciaMetros = calcularDistanciaMetros(
    latitud,
    longitud,
    latitudSucursal,
    longitudSucursal
  );

  const precisionMaxima = Number(
    configuracion.precisionMaximaMetros
  );

  const precisionAceptable = precision <= precisionMaxima;

  const dentroDelRadio = distanciaMetros <= radioMetros;

  const dentroConMargenDePrecision =
    distanciaMetros + precision <= radioMetros;

  return {
    distanciaMetros: Math.round(distanciaMetros),
    radioMetros,
    precisionMetros: Math.round(precision),
    precisionAceptable,
    dentroDelRadio,
    dentroConMargenDePrecision,
    autorizada:
      precisionAceptable &&
      dentroConMargenDePrecision
  };
}

function solicitarUbicacion(opciones = {}) {
  const {
    timeout = 20000,
    maximumAge = 0,
    enableHighAccuracy = true
  } = opciones;

  return new Promise((resolve, reject) => {
    if (!window.isSecureContext) {
      reject(
        crearError(
          "ubicacion/contexto-inseguro",
          "Para registrar asistencia debes abrir la aplicación mediante HTTPS."
        )
      );
      return;
    }

    if (!navigator.geolocation) {
      reject(
        crearError(
          "ubicacion/no-disponible",
          "Este dispositivo no permite consultar su ubicación."
        )
      );
      return;
    }

    navigator.geolocation.getCurrentPosition(
      posicion => {
        resolve({
          latitud: posicion.coords.latitude,
          longitud: posicion.coords.longitude,
          precision: posicion.coords.accuracy,
          altitud: posicion.coords.altitude,
          capturadaEn: posicion.timestamp
        });
      },
      error => {
        const mensajes = {
          1: "Debes permitir el acceso a tu ubicación para registrar asistencia.",
          2: "No pudimos determinar tu ubicación. Activa el GPS e inténtalo nuevamente.",
          3: "La ubicación tardó demasiado. Inténtalo nuevamente."
        };

        reject(
          crearError(
            "ubicacion/error",
            mensajes[error.code] ||
              "No fue posible obtener tu ubicación."
          )
        );
      },
      {
        timeout,
        maximumAge,
        enableHighAccuracy
      }
    );
  });
}

async function obtenerUbicacionVerificada(sucursal, configuracion) {
  const ubicacion = await solicitarUbicacion();

  const resultado = verificarGeocerca(
    ubicacion,
    sucursal,
    configuracion
  );

  if (!resultado.precisionAceptable) {
    throw crearError(
      "ubicacion/poca-precision",
      `La precisión actual es de aproximadamente ${resultado.precisionMetros} metros. Intenta acercarte a una ventana o activar la ubicación precisa.`
    );
  }

  if (!resultado.autorizada) {
    throw crearError(
      "ubicacion/fuera-de-rango",
      `No se pudo confirmar que estés dentro del área autorizada. Distancia estimada: ${resultado.distanciaMetros} metros.`
    );
  }

  return {
    ...ubicacion,
    verificacion: resultado
  };
}

function obtenerDatosUbicacion(ubicacion) {
  if (!ubicacion) {
    return null;
  }

  return {
    latitud: ubicacion.latitud,
    longitud: ubicacion.longitud,
    precision: ubicacion.precision,
    capturadaEn: ubicacion.capturadaEn
  };
}

async function prepararRegistro(empleado) {
  const usuario = obtenerUsuario();

  if (empleado?.uid !== usuario.uid) {
    throw crearError(
      "empleado/usuario-invalido",
      "La cuenta actual no corresponde al empleado seleccionado."
    );
  }

  if (!navigator.onLine) {
    throw crearError(
      "asistencia/sin-conexion",
      "Necesitas conexión a internet para registrar asistencia."
    );
  }

  if (!empleado?.uid || empleado.activo !== true) {
    throw crearError(
      "empleado/no-autorizado",
      "Tu perfil no está autorizado para registrar asistencia."
    );
  }

  const configuracion = await obtenerConfiguracionGeneral();
  const sucursal = await obtenerSucursal(empleado.sucursalId);
  const horario = await obtenerHorarioEmpleado(empleado);

  let ubicacion = null;

  if (configuracion.ubicacionObligatoria) {
    ubicacion = await obtenerUbicacionVerificada(
      sucursal,
      configuracion
    );
  }

  return {
    configuracion,
    sucursal,
    horario,
    ubicacion
  };
}

async function registrarEntrada(empleado) {
  const datos = await prepararRegistro(empleado);

  const respuesta = await registrarEntradaRemota({
    ubicacion: obtenerDatosUbicacion(datos.ubicacion)
  });

  return respuesta.data;
}

async function registrarSalida(empleado) {
  const datos = await prepararRegistro(empleado);

  const respuesta = await registrarSalidaRemota({
    ubicacion: obtenerDatosUbicacion(datos.ubicacion)
  });

  return respuesta.data;
}

function normalizarAsistencia(documento) {

    ...documento.data()
  };
}

async function obtenerJornadaActiva(uid) {
  const usuario = obtenerUsuario();

  if (uid !== usuario.uid) {
    throw crearError(
      "asistencia/usuario-invalido",
      "Solo puedes consultar tu propia jornada."
    );
  }

  const consulta = query(
    collection(db, "asistencias"),
    where("empleadoId", "==", uid),
    where("estadoJornada", "==", "en_curso"),
    limit(2)
  );

  const resultado = await getDocs(consulta);

  if (resultado.size > 1) {
    throw crearError(
      "asistencia/jornadas-duplicadas",
      "Se encontraron jornadas activas duplicadas. Comunícate con el administrador."
    );
  }

  if (resultado.empty) {
    return null;
  }

  return normalizarAsistencia(resultado.docs[0]);
}

async function obtenerUltimasAsistencias(uid, cantidad = 5) {
  const usuario = obtenerUsuario();

  if (uid !== usuario.uid) {
    throw crearError(
      "asistencia/usuario-invalido",
      "Solo puedes consultar tus propias asistencias."
    );
  }

  const consulta = query(
    collection(db, "asistencias"),
    where("empleadoId", "==", uid),
    orderBy("fechaInicioProgramada", "desc"),
    limit(Math.min(Math.max(1, cantidad), 50))
  );

  const resultado = await getDocs(consulta);

  return resultado.docs.map(normalizarAsistencia);
}

async function obtenerAsistenciasMes(uid, mes) {
  const usuario = obtenerUsuario();

  if (uid !== usuario.uid) {
    throw crearError(
      "asistencia/usuario-invalido",
      "Solo puedes consultar tus propias asistencias."
    );
  }

  if (!/^\d{4}-(0[1-9]|1[0-2])$/.test(mes)) {
    throw crearError(
      "asistencia/mes-invalido",
      "Selecciona un mes válido."
    );
  }

  const inicio = `${mes}-01`;
  const [anio, numeroMes] = mes.split("-").map(Number);
  const siguiente = new Date(Date.UTC(anio, numeroMes, 1));
  const fin = siguiente.toISOString().slice(0, 10);

  const consulta = query(
    collection(db, "asistencias"),
    where("empleadoId", "==", uid),
    where("fechaTurno", ">=", inicio),
    where("fechaTurno", "<", fin),
    orderBy("fechaTurno", "desc")
  );

  const resultado = await getDocs(consulta);

  return resultado.docs.map(normalizarAsistencia);
}

function calcularHorasTrabajadas(asistencia) {
  const entrada = convertirFecha(asistencia?.entrada);
  const salida = convertirFecha(asistencia?.salida);

  if (!entrada || !salida) {
    return 0;
  }

  const diferencia = salida.getTime() - entrada.getTime();

  if (diferencia <= 0) {
    return 0;
  }

  return diferencia / 3600000;
}

function calcularResumen(asistencias = []) {
  const resumen = {
    asistencias: 0,
    retardos: 0,
    faltas: 0,
    justificadas: 0,
    descansos: 0,
    horasTrabajadas: 0
  };

  asistencias.forEach(asistencia => {
    const estado = asistencia.estadoAsistencia;

    if (
      estado === ESTADOS.PUNTUAL ||
      estado === ESTADOS.RETARDO
    ) {
      resumen.asistencias++;
    }

    if (estado === ESTADOS.RETARDO) {
      resumen.retardos++;
    }

    if (estado === ESTADOS.FALTA) {
      resumen.faltas++;
    }

    if (estado === ESTADOS.JUSTIFICADA) {
      resumen.justificadas++;
    }

    if (estado === ESTADOS.DESCANSO) {
      resumen.descansos++;
    }

    resumen.horasTrabajadas += calcularHorasTrabajadas(
      asistencia
    );
  });

  resumen.horasTrabajadas = Math.round(
    resumen.horasTrabajadas * 100
  ) / 100;

  return resumen;
}

function crearElementoHistorial(asistencia) {
  const item = document.createElement("div");
  item.className = "historial-item";

  const informacion = document.createElement("div");
  informacion.className = "historial-item-informacion";

  const icono = document.createElement("div");
  icono.className = "historial-item-icono";
  icono.textContent = "✓";

  const textos = document.createElement("div");
  textos.className = "historial-item-textos";

  const fecha = document.createElement("strong");
  fecha.textContent = formatearFecha(
    obtenerFechaRegistro(asistencia),
    {
      day: "numeric",
      month: "short",
      year: "numeric"
    }
  );

  const estado = document.createElement("small");
  estado.textContent = obtenerNombreEstado(
    asistencia.estadoAsistencia ||
    asistencia.estadoJornada
  );

  textos.append(fecha, estado);
  informacion.append(icono, textos);

  const datos = document.createElement("div");
  datos.className = "historial-item-datos";

  const horas = document.createElement("strong");
  horas.textContent =
    `${formatearHora(asistencia.entrada)} - ${formatearHora(asistencia.salida)}`;

  const estadoJornada = document.createElement("small");
  estadoJornada.textContent =
    asistencia.estadoJornada === "en_curso"
      ? "En curso"
      : "Registro";

  datos.append(horas, estadoJornada);
  item.append(informacion, datos);

  return item;
}

function renderizarHistorial(contenedorId, registros) {
  const contenedor = $(contenedorId);

  if (!contenedor) {
    return;
  }

  contenedor.replaceChildren();

  if (!registros.length) {
    contenedor.append(
      crearHistorialVacio(
        "Sin registros",
        "Todavía no hay asistencias para mostrar."
      )
    );
    return;
  }

  const fragmento = document.createDocumentFragment();

  registros.forEach(registro => {
    fragmento.append(crearElementoHistorial(registro));
  });

  contenedor.append(fragmento);
}

function abrirModal(id) {
  const modal = $(id);

  if (!modal) {
    return;
  }

  if (modalActivo) {
    cerrarModal(modalActivo);
  }

  modal.hidden = false;
  modalActivo = id;
  document.body.style.overflow = "hidden";
}

function cerrarModal(id) {
  const modal = $(id);

  if (modal) {
    modal.hidden = true;
  }

  if (modalActivo === id) {
    modalActivo = null;
  }

  if (!modalActivo) {
    document.body.style.overflow = "";
  }

  if (id === "modalConfirmacion") {
    operacionActual = null;
  }
}

function prepararConfirmacion(tipo) {
  if (
    !empleado ||
    procesandoRegistro ||
    cargandoDatos ||
    !navigator.onLine
  ) {
    return;
  }

  const entrada = tipo === "entrada";

  if (entrada && jornadaActual) {
    mostrarToast(
      "Ya tienes una jornada activa.",
      "error"
    );
    return;
  }

  if (!entrada && !jornadaActual) {
    mostrarToast(
      "No tienes una jornada activa para finalizar.",
      "error"
    );
    return;
  }

  operacionActual = tipo;

  establecerTexto(
    "tituloConfirmacion",
    entrada ? "Registrar entrada" : "Registrar salida"
  );

  establecerTexto(
    "descripcionConfirmacion",
    entrada
      ? "¿Confirmas que deseas comenzar tu jornada? Verificaremos tu ubicación y horario."
      : "¿Confirmas que has terminado tu jornada? Verificaremos tu ubicación antes de registrar la salida."
  );

  establecerTexto(
    "confirmacionIcono",
    entrada ? "✓" : "↗"
  );

  const boton = $("aceptarConfirmacion");

  if (boton) {
    boton.disabled = false;
    boton.textContent = entrada
      ? "Confirmar entrada"
      : "Confirmar salida";
  }

  actualizarReloj();
  abrirModal("modalConfirmacion");
}

async function confirmarRegistro() {
  if (!operacionActual || procesandoRegistro || !empleado) {
    return;
  }

  const tipo = operacionActual;
  const boton = $("aceptarConfirmacion");

  procesandoRegistro = true;

  if (boton) {
    boton.disabled = true;
    boton.textContent = "Verificando...";
  }

  actualizarBotones();
  limpiarMensajeAsistencia();

  try {
    const resultado = tipo === "entrada"
      ? await registrarEntrada(empleado)
      : await registrarSalida(empleado);

    cerrarModal("modalConfirmacion");

    const mensaje = resultado?.mensaje ||
      (tipo === "entrada"
        ? "Tu entrada fue registrada correctamente."
        : "Tu salida fue registrada correctamente.");

    mostrarMensajeAsistencia(mensaje);
    mostrarToast(mensaje);

    procesandoRegistro = false;

    await cargarDatos();
  } catch (error) {
    console.error("Error al registrar asistencia:", error);

    const mensaje = obtenerError(error);

    mostrarMensajeAsistencia(mensaje, "error");
    mostrarToast(mensaje, "error");

    if (boton) {
      boton.textContent = tipo === "entrada"
        ? "Confirmar entrada"
        : "Confirmar salida";
    }
  } finally {
    procesandoRegistro = false;

    if (boton) {
      boton.disabled = false;
    }

    actualizarBotones();
  }
}

async function abrirHistorial() {

    );
  }
}

function configurarModales() {
  $("botonEntrada")?.addEventListener("click", () => {
    prepararConfirmacion("entrada");
  });

  $("botonSalida")?.addEventListener("click", () => {
    prepararConfirmacion("salida");
  });

  $("aceptarConfirmacion")?.addEventListener(
    "click",
    confirmarRegistro
  );

  $("cancelarConfirmacion")?.addEventListener("click", () => {
    if (!procesandoRegistro) {
      cerrarModal("modalConfirmacion");
    }
  });

  $("cerrarModalPerfil")?.addEventListener("click", () => {
    cerrarModal("modalPerfil");
  });

  $("cerrarModalHistorial")?.addEventListener("click", () => {
    cerrarModal("modalHistorial");
  });

  document.querySelectorAll("[data-cerrar-modal]").forEach(fondo => {
    fondo.addEventListener("click", () => {
      if (procesandoRegistro) {
        return;
      }

      const tipo = fondo.dataset.cerrarModal;

      const ids = {
        confirmacion: "modalConfirmacion",
        perfil: "modalPerfil",
        historial: "modalHistorial"
      };

      if (ids[tipo]) {
        cerrarModal(ids[tipo]);
      }
    });
  });

  document.addEventListener("keydown", evento => {
    if (evento.key === "Escape" && modalActivo && !procesandoRegistro) {
      cerrarModal(modalActivo);
    }
  });

  $("filtroMesHistorial")?.addEventListener(
    "change",
    cargarHistorialCompleto
  );
}

function configurarSesion() {

function configurarEventos() {

    actualizarConexion();

  document.addEventListener("visibilitychange", () => {
    if (!document.hidden) {
      actualizarReloj();
      actualizarConexion();
      cargarDatos();
    }
  });

  window.addEventListener("pagehide", () => {
    clearInterval(relojIntervalo);
    clearInterval(actualizacionIntervalo);
    clearTimeout(temporizadorToast);
  });
}

function iniciarActualizacionPeriodica() {

    if (
      !document.hidden &&
      navigator.onLine &&
      !procesandoRegistro
    ) {
      cargarDatos();
    }
  }, 60000);
}

function convertirFecha(valor) {
  if (!valor) {
    return null;
  }

  if (valor instanceof Date) {
    return Number.isNaN(valor.getTime()) ? null : valor;
  }

  if (typeof valor.toDate === "function") {
    return convertirFecha(valor.toDate());
  }

  if (typeof valor.seconds === "number") {
    return convertirFecha(new Date(valor.seconds * 1000));
  }

  if (typeof valor === "number") {
    return convertirFecha(new Date(valor));
  }

  if (typeof valor === "string") {
    return convertirFecha(new Date(valor));
  }

  return null;
}

function obtenerFechaLocal(fecha = new Date()) {
  const partes = FORMATO_DIA.formatToParts(fecha);
  const datos = Object.fromEntries(
    partes.map(parte => [parte.type, parte.value])
  );

  return `${datos.year}-${datos.month}-${datos.day}`;
}

function formatearFecha(valor) {

}

}

  return String(valor ?? "")
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .toLocaleLowerCase("es-MX")
    .trim();
}

function obtenerNombreEmpleado(registro) {
  if (registro.nombreCompleto) {
    return registro.nombreCompleto;
  }

  if (registro.nombreEmpleado) {
    return registro.nombreEmpleado;
  }

  const empleado = empleados.find(
    item => item.uid === registro.empleadoId
  );

  return empleado?.nombreCompleto || "Empleado no identificado";
}

function obtenerIniciales(nombre) {
  const partes = String(nombre)
    .trim()
    .split(/\s+/)
    .filter(Boolean);

  if (!partes.length) {
    return "EM";
  }

  return partes.length === 1
    ? partes[0].slice(0, 2).toUpperCase()
    : `${partes[0][0]}${partes[1][0]}`.toUpperCase();
}

function obtenerEntrada(registro) {
  return convertirFecha(
    registro.entrada?.fechaHora ??
    registro.entrada?.timestamp ??
    registro.fechaEntrada ??
    registro.horaEntradaReal ??
    registro.entrada
  );
}

function obtenerSalida(registro) {
  return convertirFecha(
    registro.salida?.fechaHora ??
    registro.salida?.timestamp ??
    registro.fechaSalida ??
    registro.horaSalidaReal ??
    registro.salida
  );
}

function obtenerMinutosTrabajados(registro) {
  const entrada = obtenerEntrada(registro);
  const salida = obtenerSalida(registro);

  if (!entrada || !salida) {
    return null;
  }

  const minutos = Math.round(
    (salida.getTime() - entrada.getTime()) / 60000
  );

  if (minutos < 0 || minutos > 24 * 60) {
    return null;
  }

  return minutos;
}

function formatearDuracion(minutos) {
  if (!Number.isFinite(minutos)) {
    return "—";
  }

  const horas = Math.floor(minutos / 60);
  const restantes = minutos % 60;

  return `${horas} h ${String(restantes).padStart(2, "0")} min`;
}

function obtenerEstado(registro) {
  const estado = normalizarTexto(
    registro.estadoAsistencia ??
    registro.estado ??
    registro.clasificacion ??
    ""
  );

  if (["falta", "ausente"].includes(estado)) {
    return {
      texto: "Falta",
      clase: "roja",
      clave: "falta"
    };
  }

  if (["retardo", "tarde", "retardo_grave"].includes(estado)) {
    return {
      texto: "Retardo",
      clase: "amarilla",
      clave: "retardo"
    };
  }

  if (["puntual", "a_tiempo", "atiempo"].includes(estado)) {
    return {
      texto: "Puntual",
      clase: "verde",
      clave: "puntual"
    };
  }

  if (["justificada", "justificado", "permiso"].includes(estado)) {
    return {
      texto: "Justificado",
      clase: "azul",
      clave: "justificado"
    };
  }

  if (["fuera_de_horario", "rechazada"].includes(estado)) {
    return {
      texto: "Fuera de horario",
      clase: "roja",
      clave: "fuera_de_horario"
    };
  }

  if (obtenerEntrada(registro) && !obtenerSalida(registro)) {
    return {
      texto: "Jornada abierta",
      clase: "azul",
      clave: "abierta"
    };
  }

  if (obtenerEntrada(registro) && obtenerSalida(registro)) {
    return {
      texto: "Completada",
      clase: "verde",
      clave: "completada"
    };
  }

  return {
    texto: "Sin clasificar",
    clase: "gris",
    clave: "desconocido"
  };
}

function obtenerFechaJornada(registro) {
  const fecha = registro.fechaJornada ??
    registro.fecha ??
    registro.dia;

  if (typeof fecha === "string" && /^\d{4}-\d{2}-\d{2}$/.test(fecha)) {
    return fecha;
  }

  const entrada = obtenerEntrada(registro);

  return entrada ? obtenerFechaLocal(entrada) : null;
}

function establecerMensajeTabla(mensaje) {
  const cuerpo = $("tablaAsistencias");

  if (!cuerpo) {
    return;
  }

  const fila = elemento("tr");
  const celda = elemento(
    "td",
    "admin-tabla-vacia",
    mensaje
  );

  celda.colSpan = 7;
  fila.append(celda);
  cuerpo.replaceChildren(fila);
}

function crearCeldaEmpleado(registro) {
  const nombre = obtenerNombreEmpleado(registro);
  const celda = elemento("td");
  const usuario = elemento("div", "admin-tabla-usuario");

  const avatar = elemento(
    "div",
    "admin-tabla-avatar",
    obtenerIniciales(nombre)
  );

  const informacion = elemento("div");

  const titulo = elemento("strong", "", nombre);
  const subtitulo = elemento(
    "small",
    "",
    registro.puesto ||
    registro.rol ||
    "Personal"
  );

  informacion.append(titulo, subtitulo);
  usuario.append(avatar, informacion);
  celda.append(usuario);

  return celda;
}

function crearFilaAsistencia(registro) {
  const fila = elemento("tr");
  const estado = obtenerEstado(registro);

  const fecha = obtenerFechaJornada(registro);
  const entrada = obtenerEntrada(registro);
  const salida = obtenerSalida(registro);
  const minutos = obtenerMinutosTrabajados(registro);

  const celdaFecha = elemento(
    "td",
    "",
    fecha || "—"
  );

  const celdaEntrada = elemento(
    "td",
    "",
    formatearHora(entrada)
  );

  const celdaSalida = elemento(
    "td",
    "",
    formatearHora(salida)
  );

  const celdaHoras = elemento(
    "td",
    "",
    formatearDuracion(minutos)
  );

  const celdaEstado = elemento("td");

  celdaEstado.append(
    elemento(
      "span",
      `admin-insignia ${estado.clase}`,
      estado.texto
    )
  );

  fila.append(
    crearCeldaEmpleado(registro),
    celdaFecha,
    celdaEntrada,
    celdaSalida,
    celdaHoras,
    celdaEstado
  );

  const celdaDetalle = elemento("td");

  const boton = elemento(
    "button",
    "admin-accion-tabla",
    "Detalles"
  );

  boton.type = "button";
  boton.addEventListener("click", () => {
    mostrarDetalleAsistencia(registro);
  });

  celdaDetalle.append(boton);
  fila.append(celdaDetalle);

  return fila;
}

function obtenerFiltros() {
  return {
    fecha: $("filtroFechaAsistencias")?.value || "",
    empleado: $("filtroEmpleadoAsistencias")?.value || "",
    estado: $("filtroEstadoAsistencias")?.value || ""
  };
}

function filtrarAsistencias() {
  const filtros = obtenerFiltros();

  return asistencias.filter(registro => {
    const coincideFecha =
      !filtros.fecha ||
      obtenerFechaJornada(registro) === filtros.fecha;

    const coincideEmpleado =
      !filtros.empleado ||
      registro.empleadoId === filtros.empleado;

    const estado = obtenerEstado(registro);

    const coincideEstado =
      !filtros.estado ||
      estado.clave === filtros.estado;

    return coincideFecha &&
      coincideEmpleado &&
      coincideEstado;
  });
}

function renderizarAsistencias() {
  const cuerpo = $("tablaAsistencias");

  if (!cuerpo) {
    return;
  }

  const resultados = filtrarAsistencias();

  if (!resultados.length) {
    establecerMensajeTabla(
      asistencias.length
        ? "No hay asistencias que coincidan con los filtros."
        : "No se encontraron asistencias registradas."
    );
    return;
  }

  const fragmento = document.createDocumentFragment();

  resultados.forEach(registro => {
    fragmento.append(
      crearFilaAsistencia(registro)
    );
  });

  cuerpo.replaceChildren(fragmento);
}

function actualizarOpcionesEmpleados() {
  const selector = $("filtroEmpleadoAsistencias");

  if (!selector) {
    return;
  }

  const seleccionAnterior = selector.value;

  selector.replaceChildren(
    new Option("Todos los empleados", "")
  );

  empleados.forEach(empleado => {
    selector.add(
      new Option(
        empleado.nombreCompleto || empleado.uid,
        empleado.uid
      )
    );
  });

  if ([...selector.options].some(
    opcion => opcion.value === seleccionAnterior
  )) {
    selector.value = seleccionAnterior;
  }
}

async function cargarEmpleados() {
  const resultado = await getDocs(
    collection(db, "empleados")
  );

  empleados = resultado.docs.map(documento => ({
    ...documento.data(),
    uid: documento.id
  }));

  empleados.sort((a, b) => {
    return String(a.nombreCompleto || "").localeCompare(
      String(b.nombreCompleto || ""),
      "es-MX"
    );
  });

  actualizarOpcionesEmpleados();
}

async function cargarAsistencias() {
  if (cargando) {
    return;
  }

  cargando = true;
  establecerMensajeTabla("Consultando asistencias...");

  try {
    const resultado = await getDocs(
      query(
        collection(db, "asistencias"),
        orderBy("fechaJornada", "desc"),
        limit(LIMITE_CONSULTA)
      )
    );

    asistencias = resultado.docs.map(documento => ({
      ...documento.data(),
      id: documento.id
    }));

    ultimaConsulta = new Date();

    renderizarAsistencias();
    actualizarMetricasDashboard();
    renderizarActividadReciente();
  } catch (error) {
    console.error("Error al cargar asistencias:", error);

    establecerMensajeTabla(
      "No fue posible consultar las asistencias."
    );

    mostrarToast(
      "No se pudieron cargar las asistencias. Revisa tu conexión y los permisos de Firebase.",
      "error"
    );
  } finally {
    cargando = false;
  }
}

function obtenerAsistenciasHoy() {
  const hoy = obtenerFechaLocal();

  return asistencias.filter(
    registro => obtenerFechaJornada(registro) === hoy
  );
}

function actualizarMetricasDashboard() {
  const registrosHoy = obtenerAsistenciasHoy();

  const empleadosPresentes = new Set(
    registrosHoy
      .filter(registro => obtenerEntrada(registro))
      .map(registro => registro.empleadoId)
      .filter(Boolean)
  );

  const retardos = registrosHoy.filter(
    registro => obtenerEstado(registro).clave === "retardo"
  ).length;

  const jornadasCompletadas = registrosHoy.filter(
    registro => obtenerEntrada(registro) && obtenerSalida(registro)
  ).length;

  if ($("metricaAsistencias")) {
    $("metricaAsistencias").textContent =
      empleadosPresentes.size;
  }

  if ($("metricaRetardos")) {
    $("metricaRetardos").textContent =
      retardos;
  }

  if ($("metricaJornadas")) {
    $("metricaJornadas").textContent =
      jornadasCompletadas;
  }
}

function crearActividad(registro) {
  const nombre = obtenerNombreEmpleado(registro);
  const estado = obtenerEstado(registro);

  const contenedor = elemento(
    "div",
    "admin-auditoria-item"
  );

  const icono = elemento(
    "div",
    "admin-auditoria-icono",
    obtenerSalida(registro) ? "✓" : "↗"
  );

  const informacion = elemento(
    "div",
    "admin-auditoria-texto"
  );

  const titulo = elemento(
    "strong",
    "",
    nombre
  );

  const descripcion = elemento(
    "p",
    "",
    `${estado.texto} · Entrada: ${formatearHora(obtenerEntrada(registro))} · Salida: ${formatearHora(obtenerSalida(registro))}`
  );

  informacion.append(titulo, descripcion);

  const hora = elemento(
    "span",
    "admin-auditoria-fecha",
    formatearHora(
      obtenerSalida(registro) || obtenerEntrada(registro)
    )
  );

  contenedor.append(icono, informacion, hora);

  return contenedor;
}

function renderizarActividadReciente() {
  const contenedor = $("dashboardActividad");

  if (!contenedor) {
    return;
  }

  const registros = obtenerAsistenciasHoy()
    .sort((a, b) => {
      const fechaA = obtenerSalida(a) || obtenerEntrada(a);
      const fechaB = obtenerSalida(b) || obtenerEntrada(b);

      return (fechaB?.getTime() || 0) -
        (fechaA?.getTime() || 0);
    })
    .slice(0, 6);

  if (!registros.length) {
    const vacio = elemento(
      "div",
      "admin-estado-vacio"
    );

    vacio.append(
      elemento(
        "strong",
        "",
        "Todavía no hay movimientos"
      ),
      elemento(
        "p",
        "",
        "Las entradas y salidas de hoy aparecerán aquí."
      )
    );

    contenedor.replaceChildren(vacio);
    return;
  }

  const fragmento = document.createDocumentFragment();

  registros.forEach(registro => {
    fragmento.append(crearActividad(registro));
  });

  contenedor.replaceChildren(fragmento);
}

function mostrarDetalleAsistencia(registro) {
  const estado = obtenerEstado(registro);
  const nombre = obtenerNombreEmpleado(registro);

  const datos = [
    `Empleado: ${nombre}`,
    `Fecha de jornada: ${obtenerFechaJornada(registro) || "—"}`,
    `Entrada: ${formatearFecha(obtenerEntrada(registro))} ${formatearHora(obtenerEntrada(registro))}`,
    `Salida: ${formatearFecha(obtenerSalida(registro))} ${formatearHora(obtenerSalida(registro))}`,
    `Estado: ${estado.texto}`,
    `Tiempo trabajado: ${formatearDuracion(obtenerMinutosTrabajados(registro))}`,
    `Identificador: ${registro.id}`
  ];

  window.alert(datos.join("\n"));
}

function configurarFiltros() {
  [
    "filtroFechaAsistencias",
    "filtroEmpleadoAsistencias",
    "filtroEstadoAsistencias"
  ].forEach(id => {
    const control = $(id);

    control?.addEventListener(
      id === "filtroFechaAsistencias" ? "change" : "input",
      renderizarAsistencias
    );
  });
}

function configurarEventos() {
  $("botonActualizarAsistencias")?.addEventListener(
    "click",
    cargarAsistencias
  );

  document.addEventListener("admin:seccion", evento => {
    if (evento.detail?.seccion === "asistencias") {
      cargarAsistencias();
    }

    if (evento.detail?.seccion === "dashboard") {
      actualizarMetricasDashboard();
      renderizarActividadReciente();
    }
  });
}

async function inicializarAsistencias(dependencias = {}) {
  if (inicializado) {
    return;
  }

  mostrarToast =
    dependencias.mostrarToast || mostrarToast;

  inicializado = true;

  configurarFiltros();
  configurarEventos();

  try {
    await cargarEmpleados();
  } catch (error) {
    console.error("Error al cargar empleados:", error);

    mostrarToast(
      "No fue posible cargar el catálogo de empleados.",
      "error"
    );
  }

  await cargarAsistencias();
}

export {
  inicializarAsistencias,
  cargarAsistencias,
  renderizarAsistencias,
  actualizarMetricasDashboard,
  renderizarActividadReciente,
  obtenerAsistenciasHoy,
  obtenerEstado,
  obtenerMinutosTrabajados
};
